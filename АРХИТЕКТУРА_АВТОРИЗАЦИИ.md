### **4.2. AuthManager (модуль авторизации)**

```python
# modules/auth/auth_manager.py

import jwt
import json
from pathlib import Path
from datetime import datetime, timedelta
from typing import Optional, Dict
import requests
from loguru import logger

class AuthManager:
    """Управление JWT токенами и авторизацией"""
    
    TOKEN_FILE = Path.home() / ".hintsage" / "token.jwt"
    REFRESH_FILE = Path.home() / ".hintsage" / "refresh.jwt"
    PUBLIC_KEY_FILE = Path.home() / ".hintsage" / "public_key.pem"
    
    API_BASE = "https://api.hintsage.com"  # TODO: заменить на реальный
    
    def __init__(self):
        self.access_token: Optional[str] = None
        self.refresh_token: Optional[str] = None
        self.features: Dict = {}
        
        # Создаем директорию если нет
        self.TOKEN_FILE.parent.mkdir(exist_ok=True)
        
        # Загружаем публичный ключ для offline валидации
        self._load_public_key()
    
    def _load_public_key(self):
        """Загружает публичный ключ для проверки JWT"""
        if not self.PUBLIC_KEY_FILE.exists():
            # Скачиваем с сервера
            try:
                resp = requests.get(f"{self.API_BASE}/auth/public-key")
                self.PUBLIC_KEY_FILE.write_text(resp.text)
            except Exception as e:
                logger.error(f"Не удалось загрузить публичный ключ: {e}")
                return
        
        self.public_key = self.PUBLIC_KEY_FILE.read_text()
    
    def is_authenticated(self) -> bool:
        """Проверяет наличие валидного токена"""
        # 1. Проверяем файл
        if not self.TOKEN_FILE.exists():
            return False
        
        # 2. Загружаем токен
        self.access_token = self.TOKEN_FILE.read_text().strip()
        
        # 3. Проверяем offline (signature + expiration)
        try:
            payload = jwt.decode(
                self.access_token,
                self.public_key,
                algorithms=["RS256"]
            )
            
            # Проверяем expiration
            exp = payload.get("exp")
            if exp and datetime.utcfromtimestamp(exp) < datetime.utcnow():
                logger.warning("Токен истек")
                return False
            
            # Загружаем features
            self.features = payload.get("features", {})
            
            logger.info(f"✅ Авторизован как {payload.get('email')} (tier: {payload.get('tier')})")
            return True
            
        except jwt.InvalidSignatureError:
            logger.error("❌ Неверная подпись токена")
            return False
        except jwt.ExpiredSignatureError:
            logger.warning("⏰ Токен истек, пробуем refresh...")
            return self._try_refresh()
        except Exception as e:
            logger.error(f"❌ Ошибка проверки токена: {e}")
            return False
    
    def login(self, email: str, password: str) -> bool:
        """Выполняет вход и получает JWT"""
        try:
            resp = requests.post(
                f"{self.API_BASE}/auth/login",
                json={"email": email, "password": password},
                timeout=10
            )
            
            if resp.status_code != 200:
                logger.error(f"❌ Ошибка входа: {resp.json()}")
                return False
            
            data = resp.json()
            
            # Сохраняем токены
            self.access_token = data["access_token"]
            self.refresh_token = data.get("refresh_token")
            self.features = data.get("features", {})
            
            self.TOKEN_FILE.write_text(self.access_token)
            if self.refresh_token:
                self.REFRESH_FILE.write_text(self.refresh_token)
            
            logger.info(f"✅ Успешный вход: {email}")
            return True
            
        except Exception as e:
            logger.error(f"❌ Ошибка при входе: {e}")
            return False
    
    def _try_refresh(self) -> bool:
        """Пытается обновить access token используя refresh token"""
        if not self.REFRESH_FILE.exists():
            return False
        
        refresh_token = self.REFRESH_FILE.read_text().strip()
        
        try:
            resp = requests.post(
                f"{self.API_BASE}/auth/refresh",
                json={"refresh_token": refresh_token},
                timeout=10
            )
            
            if resp.status_code != 200:
                return False
            
            data = resp.json()
            self.access_token = data["access_token"]
            self.TOKEN_FILE.write_text(self.access_token)
            
            logger.info("✅ Токен обновлен")
            return True
            
        except Exception as e:
            logger.error(f"❌ Ошибка refresh: {e}")
            return False
    
    def logout(self):
        """Выход (удаляет токены)"""
        self.TOKEN_FILE.unlink(missing_ok=True)
        self.REFRESH_FILE.unlink(missing_ok=True)
        self.access_token = None
        self.refresh_token = None
        self.features = {}
        logger.info("👋 Вышли из системы")
    
    def get_features(self) -> Dict:
        """Возвращает доступные фичи"""
        return self.features
    
    def has_feature(self, feature_name: str) -> bool:
        """Проверяет доступность фичи"""
        return self.features.get(feature_name, False)
```

### **4.3. LoginDialog (форма входа)**

```python
# modules/auth/login_dialog.py

from PyQt6.QtWidgets import (
    QDialog, QVBoxLayout, QLabel, QLineEdit, 
    QPushButton, QHBoxLayout, QMessageBox
)
from PyQt6.QtCore import Qt
from loguru import logger

class LoginDialog(QDialog):
    """Диалог авторизации (как в Cursor)"""
    
    def __init__(self, auth_manager):
        super().__init__()
        self.auth_manager = auth_manager
        self.authenticated = False
        
        self.setWindowTitle("Hintsage - Вход")
        self.setFixedSize(400, 300)
        self.setWindowFlags(Qt.WindowType.WindowStaysOnTopHint)
        
        self._setup_ui()
    
    def _setup_ui(self):
        """Создает интерфейс"""
        layout = QVBoxLayout()
        
        # Заголовок
        title = QLabel("🎯 Hintsage - AI Interview Assistant")
        title.setStyleSheet("font-size: 18px; font-weight: bold; margin: 20px;")
        title.setAlignment(Qt.AlignmentFlag.AlignCenter)
        layout.addWidget(title)
        
        # Email
        email_label = QLabel("Email:")
        self.email_input = QLineEdit()
        self.email_input.setPlaceholderText("your@email.com")
        layout.addWidget(email_label)
        layout.addWidget(self.email_input)
        
        # Password
        password_label = QLabel("Password:")
        self.password_input = QLineEdit()
        self.password_input.setPlaceholderText("••••••••")
        self.password_input.setEchoMode(QLineEdit.EchoMode.Password)
        self.password_input.returnPressed.connect(self._on_login)
        layout.addWidget(password_label)
        layout.addWidget(self.password_input)
        
        # Кнопки
        button_layout = QHBoxLayout()
        
        login_btn = QPushButton("Login")
        login_btn.clicked.connect(self._on_login)
        login_btn.setStyleSheet("background: #4CAF50; color: white; padding: 10px;")
        button_layout.addWidget(login_btn)
        
        subscribe_btn = QPushButton("Subscribe")
        subscribe_btn.clicked.connect(self._on_subscribe)
        subscribe_btn.setStyleSheet("background: #2196F3; color: white; padding: 10px;")
        button_layout.addWidget(subscribe_btn)
        
        layout.addLayout(button_layout)
        
        # Информация
        info = QLabel("Нет аккаунта? Нажмите 'Subscribe' чтобы создать")
        info.setStyleSheet("color: gray; font-size: 12px; margin-top: 10px;")
        info.setAlignment(Qt.AlignmentFlag.AlignCenter)
        layout.addWidget(info)
        
        self.setLayout(layout)
    
    def _on_login(self):
        """Обработка входа"""
        email = self.email_input.text().strip()
        password = self.password_input.text()
        
        if not email or not password:
            QMessageBox.warning(self, "Ошибка", "Заполните все поля")
            return
        
        # Попытка входа
        if self.auth_manager.login(email, password):
            self.authenticated = True
            self.accept()
        else:
            QMessageBox.critical(
                self, 
                "Ошибка входа", 
                "Неверный email или пароль.\n\n"
                "Или у вас нет активной подписки.\n"
                "Нажмите 'Subscribe' чтобы купить."
            )
    
    def _on_subscribe(self):
        """Открывает страницу подписки"""
        import webbrowser
        webbrowser.open("https://hintsage.com/pricing")  # TODO: реальный URL
        
        QMessageBox.information(
            self,
            "Подписка",
            "После оплаты вернитесь сюда и войдите с вашим email/password"
        )
```

### **4.4. FeatureManager (управление доступом)**

```python
# modules/auth/feature_manager.py

from typing import Dict
from loguru import logger

class FeatureManager:
    """Управляет доступом к функциям на основе подписки"""
    
    def __init__(self, features: Dict):
        self.features = features
    
    def can_use_ocr(self) -> bool:
        """Может ли пользователь использовать OCR"""
        return self.features.get("ocr_enabled", False)
    
    def can_use_audio_both(self) -> bool:
        """Может ли использовать режим "both" для аудио"""
        mode = self.features.get("audio_mode", "microphone")
        return mode == "both"
    
    def get_max_requests_per_day(self) -> int:
        """Максимум запросов в день"""
        return self.features.get("max_requests_per_day", 10)
    
    def get_context_window(self) -> int:
        """Размер контекстного окна"""
        return self.features.get("context_window", 3000)
    
    def get_max_tokens(self) -> int:
        """Максимум токенов для ответа"""
        return self.features.get("max_tokens", 500)
    
    def get_parallel_workers(self) -> int:
        """Количество параллельных воркеров"""
        return self.features.get("parallel_workers", 1)
    
    def can_export_history(self) -> bool:
        """Может ли экспортировать историю"""
        return self.features.get("export_history", False)
    
    def can_use_advanced_prompts(self) -> bool:
        """Может ли использовать расширенные промпты"""
        return self.features.get("advanced_prompts", False)
```

### **4.5. Интеграция в main.py**

```python
# main.py (добавить в начало)

from modules.auth import AuthManager, LoginDialog, FeatureManager

def main():
    # Qt приложение
    qt_app = QApplication(sys.argv)
    
    # 1. Проверка авторизации
    auth_manager = AuthManager()
    
    if not auth_manager.is_authenticated():
        # Показываем форму входа
        login_dialog = LoginDialog(auth_manager)
        if login_dialog.exec() != QDialog.DialogCode.Accepted:
            logger.warning("Отменен вход, выход")
            return
    
    # 2. Загружаем features
    feature_manager = FeatureManager(auth_manager.get_features())
    
    # 3. Запускаем приложение с ограничениями
    app = HintsageApp(qt_app, feature_manager)
    app.start()
    
    qt_app.exec()
```
