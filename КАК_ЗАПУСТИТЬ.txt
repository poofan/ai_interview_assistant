📌 СПОСОБЫ ЗАПУСКА:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

┌────────────────────────┬───────────────┬──────────────────────┐
│ Файл                   │ Консоль       │ Когда использовать   │
├────────────────────────┼───────────────┼──────────────────────┤
│ start_silent.vbs    ⭐ │ НЕТ (фон)     │ Интервью (невидимо)  │
│ start_background.bat   │ Закрывается   │ Обычная работа       │
│ start.bat              │ ДА (видна)    │ Отладка/тестирование │
│ start_quick.bat        │ ДА (видна)    │ Быстрый тест         │
└────────────────────────┴───────────────┴──────────────────────┘

⭐ РЕКОМЕНДУЕТСЯ: start_silent.vbs
  → Полностью невидимый запуск
  → Никаких окон консоли
  → Работает в фоне
  → Управление через горячие клавиши


📌 СОЗДАТЬ ЯРЛЫК НА РАБОЧЕМ СТОЛЕ:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. Правой кнопкой мыши на:
   create_shortcut.ps1
   
2. "Запуск от имени администратора"

3. Готово! На рабочем столе появится ярлык "Hintsage AI"


📌 ЧЕРЕЗ КОМАНДНУЮ СТРОКУ:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
cd C:\git\ai_interview_assistant
venv\Scripts\python.exe main.py


⌨️ ГОРЯЧИЕ КЛАВИШИ (работают в любом приложении):
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  Ctrl+Shift+H  →  Показать/скрыть окно
  Ctrl+Shift+Q  →  Распознать речь (последние 10 сек)
  Ctrl+Shift+S  →  Скриншот области + OCR
  Ctrl+Shift+C  →  Очистить контекст
  Ctrl+Shift+X  →  Выход
  Esc           →  Быстро скрыть окно


🎯 КАК ПОЛЬЗОВАТЬСЯ НА ИНТЕРВЬЮ:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. Запустите приложение (start.bat)
   
2. Собеседник задает вопрос → ПОДОЖДИТЕ 1-2 сек
   
3. Нажмите Ctrl+Shift+Q
   
4. Читайте ответ в overlay окне
   
5. Отвечайте своими словами


🔧 ЕСЛИ НЕ РАБОТАЕТ:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
✓ Проверьте OpenAI API ключ в config.yaml
✓ Убедитесь что venv установлен (install.bat)
✓ Смотрите логи: logs\hintsage.log


⚡ GPU УСКОРЕНИЕ АКТИВНО:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

✓ EasyOCR на GPU
✓ Скорость распознавания: ~1-2 сек
